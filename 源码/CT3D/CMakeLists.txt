cmake_minimum_required(VERSION 3.16.3)
project(CT3D)
find_package(Slicer REQUIRED)
include(${Slicer_USE_FILE})
set(MODULE_PYTHON_SCRIPTS CT3D.py)
file(GLOB library_scripts RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/CT3DLib/*.py")
list(APPEND MODULE_PYTHON_SCRIPTS ${library_scripts})
slicerMacroBuildScriptedModule(NAME CT3D SCRIPTS ${MODULE_PYTHON_SCRIPTS})
set(ct_quality "${CMAKE_CURRENT_SOURCE_DIR}/../ct part/CT3DCTLib/CT3DCTQuality.py")
configure_file("${ct_quality}" "${CMAKE_BINARY_DIR}/${Slicer_QTSCRIPTEDMODULES_LIB_DIR}/CT3DLib/CT3DCTQuality.py" COPYONLY)
install(FILES "${ct_quality}" DESTINATION "${Slicer_INSTALL_QTSCRIPTEDMODULES_LIB_DIR}/CT3DLib" COMPONENT RuntimeLibraries)
